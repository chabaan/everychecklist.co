sections:
  - title: "6 Weeks Before"
    items:
      - "Set a gift budget"
      - "Make a gift list for everyone"
      - "Book travel if visiting family"
      - "Plan the holiday menu"
      - "Order anything that ships slowly"
  - title: "3 Weeks Before"
    items:
      - "Start holiday shopping"
      - "Order or write holiday cards"
      - "Plan outfits for gatherings"
      - "Schedule a deep clean of the house"
      - "Confirm holiday plans with family"
  - title: "1 Week Before"
    items:
      - "Wrap gifts"
      - "Confirm the final guest count"
      - "Grocery shop for non-perishables"
      - "Prep freezer-friendly dishes in advance"
      - "Charge the camera or phone for photos"
  - title: "2 Days Before"
    items:
      - "Final grocery run for perishables"
      - "Plan the table and seating"
      - "Tidy guest areas and the bathroom"
      - "Prep drinks and appetizers ahead of time"
  - title: "Christmas Day"
    items:
      - "Start the turkey or main dish early"
      - "Delegate small tasks to guests"
      - "Take a few photos before the chaos starts"
      - "Take a moment to actually enjoy it"
