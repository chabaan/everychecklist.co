sections:
  - title: "For Mom: During Labor"
    items:
      - "Insurance card and ID"
      - "Hospital paperwork"
      - "Birth plan copies"
      - "Phone charger with long cable"
      - "Lip balm"
      - "Hair ties or headband"
      - "Socks with grip"
      - "Comfortable robe or labor gown"
      - "Snacks for early labor, if permitted"
      - "Offline playlist"
  - title: "For Mom: After Delivery"
    items:
      - "Comfortable going-home outfit"
      - "Nursing bra, if breastfeeding"
      - "Own toiletries"
      - "Maternity pads"
      - "High-waisted underwear"
      - "Nipple cream, if breastfeeding"
      - "Glasses, if you wear contacts"
      - "A pillow from home"
  - title: "For Baby"
    items:
      - "Going-home outfit in two sizes"
      - "Swaddle blanket"
      - "Hat"
      - "Socks or booties"
      - "Car seat, installed and checked"
      - "Diapers and wipes"
      - "A few burp cloths"
  - title: "For the Partner"
    items:
      - "Change of clothes for two days"
      - "Toiletries"
      - "Phone charger"
      - "Snacks and cash"
      - "Pillow and blanket"
      - "List of who to call"
      - "Camera or phone"
  - title: "Don't Forget"
    items:
      - "Car seat base installed and inspected"
      - "House key accessible to helpers"
      - "Feeding instructions note for anyone helping"
