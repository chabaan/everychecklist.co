sections:
  - title: "Before You Start"
    items:
      - "Plan 3-5 meals for the week, not seven"
      - "Check what's already in the fridge and pantry"
      - "Choose recipes that share ingredients"
      - "Pick one protein, one grain, one vegetable to batch"
      - "Set aside 2-3 hours for prep day"
  - title: "Grocery List Categories"
    items:
      - "Proteins (chicken, tofu, beans, eggs)"
      - "Grains and starches (rice, pasta, potatoes)"
      - "Vegetables, fresh and frozen"
      - "Sauces and seasonings"
      - "Snacks and breakfast items"
  - title: "Prep Day Steps"
    items:
      - "Wash and chop all vegetables first"
      - "Cook grains in bulk"
      - "Cook proteins in bulk"
      - "Portion meals into containers"
      - "Label containers with contents and date"
  - title: "Storage"
    items:
      - "Use airtight, stackable containers"
      - "Cool food fully before sealing"
      - "Freeze what won't be eaten within 4 days"
      - "Keep a running list of what's in the freezer"
  - title: "Reheating & Safety"
    items:
      - "Reheat to steaming hot, not just warm"
      - "Eat refrigerated meals within 3-4 days"
      - "Don't refreeze thawed meals"
      - "Keep raw and cooked items separate in the fridge"
